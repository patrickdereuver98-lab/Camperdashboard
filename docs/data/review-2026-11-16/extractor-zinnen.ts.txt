const depOf = (r: Record<string, unknown>) => (r.departure ? (r.departure === "vast" ? `vast ${r.checkout_until}` : String(r.departure)) : "-");
const FIELD = (k: string) =>
  k === "dep" ? "vertrek" : k === "checkin_from" ? "aankomst" : k.startsWith("price_") ? "prijs" : k.startsWith("extra:") || k === "tourist_tax_included" ? "toeslagen" :
  k === "season" || k === "open_all_year" ? "seizoen" : k.startsWith("access") ? "toegang" : "voorzieningen";
const show = (v: unknown): string => (v === undefined ? "leeg" : v instanceof RegExp ? String(v) : typeof v === "object" && v && "oneOf" in v ? (v as { oneOf: unknown[] }).oneOf.map(show).join(" of ") : JSON.stringify(v));
const eq = (got: unknown, exp: unknown): boolean => {
  if (exp instanceof RegExp) return typeof got === "string" && exp.test(got);
  if (exp && typeof exp === "object" && "oneOf" in exp) return (exp as { oneOf: unknown[] }).oneOf.some((x) => eq(got, x));
  return JSON.stringify(got) === JSON.stringify(exp);
};
const rows: { id: string; input: string; field: string; key: string; exp: string; got: string; ok: boolean; note?: string }[] = [];
for (const c of C) {
  const pages = Array.isArray(c.html) ? c.html : [c.html];
  const r = extract(pages, base) as Record<string, unknown>;
  const extras = (r.price_extras as { label: string; amount: number; unit?: string }[] | undefined) ?? [];
  for (const [k, exp] of Object.entries(c.exp)) {
    let got: unknown;
    let e: unknown = exp;
    if (k === "dep") got = depOf(r);
    else if (k.startsWith("extra:")) {
      const x = extras.find((y) => y.label === k.slice(6));
      got = x ? [x.amount, x.unit] : undefined;
      e = Array.isArray(exp) ? [exp[0], exp[1]] : exp;
      // JSON.stringify maakt van undefined null: gelijk trekken
      if (Array.isArray(got)) got = [got[0], got[1] ?? null];
      if (Array.isArray(e)) e = [e[0], e[1] ?? null];
    } else got = r[k];
    rows.push({
      id: c.id, input: pages.join(" ⏎ ").replace(/<[^>]+>/g, " ").replace(/\s+/g, " ").trim(), field: FIELD(k), key: k,
      exp: show(e), got: show(got), ok: eq(got, e), note: k === "dep" || k.startsWith("price") ? String(r.times_note ?? r.price_note ?? "") : "",
    });
  }
}

}
